1. Get the application URL by running these commands:
{{- if .Values.ingress.enabled }}
  {{- range .Values.ingress.hosts }}
  http{{ if $.Values.ingress.tls }}s{{ end }}://{{ .host }}{{ default "/" .path }}
  {{- end }}
{{- else }}
  export POD_NAME=$(kubectl get pods --namespace {{ .Release.Namespace }} -l \
    "app.kubernetes.io/name={{ include "deployment-app.name" . }},app.kubernetes.io/instance={{ .Release.Name }}" \
    -o name)
  echo "Port-forward to {{ .Values.service.port }}:"
  kubectl --namespace {{ .Release.Namespace }} port-forward "$POD_NAME" {{ .Values.service.port }}:{{ .Values.service.port }}
{{- end }}

2. Verify the deployment:
  kubectl rollout status deployment/{{ include "deployment-app.fullname" . }}

3. Check the health endpoint:
  curl http://localhost:{{ .Values.service.port }}/healthz
